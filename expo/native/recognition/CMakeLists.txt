cmake_minimum_required(VERSION 3.22)
project(nt_ios_recognition LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
get_filename_component(NT_REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(NT_CORE "${NT_REPO}/android/app/src/main/cpp")
# Platform-neutral core only. Never compile jni.cpp or use Android libraries.
file(GLOB_RECURSE NT_CORE_SOURCES CONFIGURE_DEPENDS "${NT_CORE}/src/*.cpp")
add_library(nt_recognition STATIC ${NT_CORE_SOURCES} TimerSession.cpp)
target_include_directories(nt_recognition PUBLIC "${NT_CORE}/include" "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_options(nt_recognition PRIVATE -O3 -fno-fast-math -ffp-contract=off -fexceptions -frtti)
find_package(Threads REQUIRED)
target_link_libraries(nt_recognition PUBLIC Threads::Threads)
if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
  enable_language(OBJCXX)
  target_sources(nt_recognition PRIVATE RecognitionBridge.mm VisionAccounts.mm)
  set_source_files_properties(RecognitionBridge.mm VisionAccounts.mm PROPERTIES COMPILE_FLAGS "-fobjc-arc")
  target_link_libraries(nt_recognition PUBLIC "-framework Foundation" "-framework CoreVideo" "-framework CoreImage" "-framework ImageIO" "-framework CoreGraphics" "-framework Vision")
endif()
if(NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
  enable_testing()
  add_executable(timer_session_test tests/TimerSessionTest.cpp)
  target_link_libraries(timer_session_test PRIVATE nt_recognition)
  target_compile_options(timer_session_test PRIVATE -UNDEBUG)
  add_test(NAME timer_session COMMAND timer_session_test)
  add_executable(ocr_memory_probe tests/OCRMemoryProbe.cpp)
  target_link_libraries(ocr_memory_probe PRIVATE nt_recognition)
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_executable(pipeline_memory_probe EXCLUDE_FROM_ALL tests/PipelineMemoryProbe.cpp)
    target_link_libraries(pipeline_memory_probe PRIVATE nt_recognition)
  endif()
endif()
